import { useState } from "react";

function App() {
  const nombre:string="Christian";
  const lenguaje:string="TypeScript";
  const experiencia:string="Ingeniero en Redes y Infraestructura";
  const experienciaProgramando:number = 1;

// const sumar = (x:number, y:number) => {

const saludar = () =>{
  console.log("Hola soy Christian en consola");
}

const lenguajeAprendiz = () => {
  console.log("Estoy aprendiendo " + lenguaje)
}

  const [contador,setContador] = useState(0);

  const aumentar = () =>{
    setContador(contador+1);
  }

  const disminuir = () => {
    setContador(contador-1);
  }

  const resetear = () => {
    setContador(0);
  }

      <button onClick={aumentar}>Aumentar</button><br></br>
      <button onClick={disminuir}>Disminuir</button><br></br>
      <button onClick={resetear}>Resetear</button>
    </div>
  )
}

export default App
